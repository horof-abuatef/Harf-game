let currentTeam = "green";
let greenScore = 0;
let redScore = 0;
let timeLeft = 30;
let timerRunning = false;

const introSound = document.getElementById("introSound");
const crowdSound = document.getElementById("crowdSound");
const countdownSound = document.getElementById("countdownSound");

const canvas = document.getElementById("hexCanvas");
const ctx = canvas.getContext("2d");

let hexItems = [];

function startGame() {
    document.getElementById("intro").style.display = "none";
    document.getElementById("game").style.display = "block";
    introSound.play();
    drawHexGrid();
}

function setTeam(team) { currentTeam = team; }
function addPoint() {
    if(crowdSound) crowdSound.play();
    if(currentTeam === "green") {
        greenScore++; document.getElementById("greenScore").textContent = greenScore;
    } else {
        redScore++; document.getElementById("redScore").textContent = redScore;
    }
}

// ===== رسم شبكة 5 صف × 6 أعمدة =====
function drawHexGrid() {
    const hexSize = 45;
    const rows = 5;
    const cols = 6;
    const startX = 120;
    const startY = 80;
    let index = 0;
    hexItems = [];

    ctx.clearRect(0,0,canvas.width,canvas.height);

    for(let row=0; row<rows; row++){
        for(let col=0; col<cols; col++){
            if(index >= letters.length) break;
            let x = startX + col*hexSize*1.8;
            let y = startY + row*hexSize*1.6;
            if(row%2===1) x += hexSize*0.9;

            drawHex(x, y, hexSize, letters[index], "white");
            hexItems.push({x, y, letter: letters[index], size: hexSize, used:false});
            index++;
        }
    }
}

function drawHex(x, y, size, letterColor, letter){
    const angle = Math.PI/3;
    ctx.fillStyle = letterColor;
    ctx.strokeStyle = "black";
    ctx.lineWidth = 2;
    ctx.beginPath();
    for(let i=0;i<6;i++){
        const px = x + size*Math.cos(angle*i);
        const py = y + size*Math.sin(angle*i);
        if(i===0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
    }
    ctx.closePath();
    ctx.fill();
    ctx.stroke();

    ctx.fillStyle="black";
    ctx.font="20px Arial";
    ctx.textAlign="center";
    ctx.textBaseline="middle";
    ctx.fillText(letter, x, y);
}

canvas.addEventListener('click', e=>{
    if(!timerRunning) return;
    const rect = canvas.getBoundingClientRect();
    const mx = e.clientX - rect.left;
    const my = e.clientY - rect.top;

    for(let hex of hexItems){
        if(!hex.used && pointInHex(mx,my,hex)){
            hex.used=true;
            addPoint();

            const color = currentTeam==="green"?"#00ff88":"#ff4444";
            redrawHex(hex,color);

            const q = questions[hex.letter] || اذكر 5 كلمات تبدأ بحرف ${hex.letter};
            alert(q);
            break;
        }
    }
});

function pointInHex(px, py, hex){
    const dx = px-hex.x;
    const dy = py-hex.y;
    const distance = Math.sqrt(dx*dx+dy*dy);
    return distance<hex.size*1.1;
}

function redrawHex(hex,color){
    const angle=Math.PI/3;
    ctx.fillStyle=color;
    ctx.beginPath();
    for(let i=0;i<6;i++){
        const px = hex.x + hex.size*Math.cos(angle*i);
        const py = hex.y + hex.size*Math.sin(angle*i);
        if(i===0) ctx.moveTo(px, py); else ctx.lineTo(px, py);
    }
    ctx.closePath();
    ctx.fill();
    ctx.stroke();

    ctx.fillStyle="black";
    ctx.font="20px Arial";
    ctx.textAlign="center";
    ctx.textBaseline="middle";
    ctx.fillText(hex.letter, hex.x, hex.y);
}

function startTimer(){
    timeLeft=30;
    timerRunning=true;
    document.getElementById("timer").textContent = timeLeft;
    countdown();
}
function countdown(){
    if(timeLeft>0){
        if(timeLeft<=5 && countdownSound) countdownSound.play();
        timeLeft--;
        document.getElementById("timer").textContent=timeLeft;
        setTimeout(countdown,1000);
    } else {
        timerRunning=false;
        announceWinner();
    }
}
function announceWinner(){
    let winner="";
    if(greenScore>redScore) winner="🟢 الفريق الأخضر فاز!";
    else if(redScore>greenScore) winner="🔴 الفريق الأحمر فاز!";
    else winner="🤝 تعادل!";
    alert(winner);
}
